cmake_minimum_required(VERSION 3.20)
project(softgraphite_layers_v10 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick DBus)
find_package(KF6CoreAddons REQUIRED)
find_package(KWin REQUIRED)
kcoreaddons_add_plugin(softgraphite_layers_v10 SOURCES layers.cpp INSTALL_NAMESPACE kwin/effects/plugins)
target_link_libraries(softgraphite_layers_v10 KWin::kwin Qt6::DBus)
