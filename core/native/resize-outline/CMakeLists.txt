cmake_minimum_required(VERSION 3.20)
project(softgraphite_outline_resize LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick DBus CorePrivate)
find_package(KF6CoreAddons REQUIRED)
find_package(KWin REQUIRED)
kcoreaddons_add_plugin(softgraphite_outline_resize SOURCES outline.cpp INSTALL_NAMESPACE kwin/effects/plugins)
target_link_libraries(softgraphite_outline_resize KWin::kwin Qt6::DBus Qt6::CorePrivate)
